121 [[nodiscard]] constexpr FC3DCell cellAt(long long index) const noexcept
122 {
123 if (index < 0 || index >= totalCellCount())
124 {
125 return FC3DCell{}; // пустая ячейка
126 }
127
128 const int nx = _gridSize.nx();
129 const int ny = _gridSize.ny();
130 const int nz = _gridSize.nz();
131
132 // 3D индексация: index = z * (nx*ny) + y * nx + x
133 const int x = static_cast<int>(index % nx);
134 const int y = static_cast<int>((index / nx) % ny);
135 const int z = static_cast<int>(index / (nx * static_cast<long long>(ny)));
136
137 const FC3DSize cs = cellSize();
138 return FC3DCell{FC3DPoint{x * cs.width(), y * cs.height(), z * cs.depth() }, cs};
139 }
