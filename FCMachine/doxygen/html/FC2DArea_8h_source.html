204 [[nodiscard]] FC2DCell at(const FC2DPoint& point) const noexcept
205 {
206 if (!contains(point))
207 {
208 return FC2DCellError;
209 }
210 FC2DSize cs = cellSize();
211 int i = static_cast<int>(std::floor((point.x() - _startPoint.x()) / cs.width()));
212 int j = static_cast<int>(std::floor((point.y() - _startPoint.y()) / cs.height()));
213 // Ограничение индексов на случай погрешностей округления на правой/нижней границе
214 i = std::min(i, cellCountX() - 1);
215 j = std::min(j, cellCountY() - 1);
216 return at(i, j);
217 }
