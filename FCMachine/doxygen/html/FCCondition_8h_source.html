382// Проверки соответствия перечислений битовым маскам
383static_assert(static_cast<uint16_t>(ErrorType::OutOfRange) <= 31, "ErrorType должен умещаться в 5 бит");
384static_assert(static_cast<uint16_t>(PanelState::ScreenSaver) <= 3, "PanelState должен умещаться в 2 бита");
385static_assert(static_cast<uint16_t>(VisibilityState::Hide) <= 1, "VisibilityState должен умещаться в 1 бит");
386static_assert(static_cast<uint16_t>(ReadyState::Error) <= 3, "ReadyState должен умещаться в 2 бита");
387static_assert(static_cast<uint16_t>(PlayState::Pause) <= 3, "PlayState должен умещаться в 2 бита");
388static_assert(static_cast<uint16_t>(ChangedState::Changed) <= 1, "ChangedState должен умещаться в 1 бит");
389
390// Проверка непересечения битовых масок (выполняется через их литеральные значения)
391static_assert((0x0003u & 0x000Cu) == 0, "READY_MASK и PLAY_MASK не должны пересекаться");
392static_assert((0x000Cu & 0x0010u) == 0, "PLAY_MASK и CHANGED_MASK не должны пересекаться");
393static_assert((0x0010u & 0x03E0u) == 0, "CHANGED_MASK и ERROR_MASK не должны пересекаться");
394static_assert((0x03E0u & 0x0C00u) == 0, "ERROR_MASK и PANEL_MASK не должны пересекаться");
395static_assert((0x0C00u & 0x1000u) == 0, "PANEL_MASK и VISIBILITY_MASK не должны пересекаться");
396
397// Проверка полного покрытия используемых битов (биты 0-12) без дыр
398static_assert((0x0003u | 0x000Cu | 0x0010u | 0x03E0u | 0x0C00u | 0x1000u) == 0x1FFFu,
399 "Битовые маски должны покрывать ровно биты 0-12 (0x1FFF)");
