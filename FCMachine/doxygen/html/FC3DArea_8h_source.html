141 [[nodiscard]] FC3DCell nextInTraversal()
142 {
143 if (_traversalOrder.isEmpty())
144 {
145 // Режим по умолчанию: последовательный обход
146 if (_traversalIndex < 0 || _traversalIndex >= totalCells())
147 {
148 return FC3DCellError;
149 }
150 FC3DCell cell = atLinear(_traversalIndex);
151 ++_traversalIndex;
152 return cell;
153 }
154 else
155 {
156 // Кастомный порядок обхода
157 if (_traversalIndex < 0 || _traversalIndex >= _traversalOrder.size())
158 {
159 return FC3DCellError;
160 }
161 int linearIdx = _traversalOrder[_traversalIndex];
162 ++_traversalIndex;
163 return atLinear(linearIdx);
164 }
165 }

196 [[nodiscard]] FC3DCell atLinear(int linearIndex) const noexcept
197 {
198 if (linearIndex < 0 || linearIndex >= totalCells())
199 {
200 return FC3DCellError;
201 }
202 int Nx = cellCountX();
203 int Ny = cellCountY();
204 int k = linearIndex / (Nx * Ny);
205 int rem = linearIndex % (Nx * Ny);
206 int j = rem / Nx;
207 int i = rem % Nx;
208 return at(i, j, k);
209 }

216 [[nodiscard]] FC3DCell at(int i, int j, int k) const noexcept
217 {
218 if (i < 0 || i >= cellCountX() ||
219 j < 0 || j >= cellCountY() ||
220 k < 0 || k >= cellCountZ())
221 {
222 return FC3DCellError;
223 }
224
225 FC3DSize cs = cellSize();
226 FC3DPoint anchor{
227 _startPoint.x() + static_cast<float>(i) * cs.width(),
228 _startPoint.y() + static_cast<float>(j) * cs.height(),
229 _startPoint.z() + static_cast<float>(k) * cs.depth()
230 };
231
232 return FC3DCell{anchor, cs};
233 }

240 [[nodiscard]] FC3DCell at(const FC3DPoint& point) const noexcept
241 {
242 if (!contains(point))
243 {
244 return FC3DCellError;
245 }
246 FC3DSize cs = cellSize();
247 int i = static_cast<int>(std::floor((point.x() - _startPoint.x()) / cs.width()));
248 int j = static_cast<int>(std::floor((point.y() - _startPoint.y()) / cs.height()));
249 int k = static_cast<int>(std::floor((point.z() - _startPoint.z()) / cs.depth()));
250 // Ограничение индексов на случай погрешностей округления на границах
251 i = std::min(i, cellCountX() - 1);
252 j = std::min(j, cellCountY() - 1);
253 k = std::min(k, cellCountZ() - 1);
254 return at(i, j, k);
255 }
